\section*{Numbers worth knowing by heart}

\begin{itemize}[itemsep=3pt]
  \item Thermal energy at room temperature: $\kB T\approx\tfrac1{40}\un{eV}$. At body temperature, $\kB T\approx4.3\un{pN\,nm}$.
  \item Energy of a visible photon: $2$--$3\un{eV}$ ($\lambda = 400$--$700\un{nm}$; use $E\,[\mathrm{eV}] = 1240/\lambda\,[\mathrm{nm}]$).
  \item Chemical bond: a few eV. Hydrogen bond: about $0.2\un{eV}$. Van der Waals: $0.01$--$0.1\un{eV}$.
  \item Nuclear binding: about $8\un{MeV}$ per nucleon. Fission: about $200\un{MeV}$ per nucleus. Fusion of hydrogen to helium: $0.7\%$ of the rest mass.
  \item Sizes: atom $10^{-10}\un{m}$, nucleus $10^{-15}\un{m}$, proton $0.84\times10^{-15}\un{m}$, cell $10^{-5}\un{m}$.
  \item Densities: water $1000\un{kg/m^3}$, air $1.2\un{kg/m^3}$, rock $3000\un{kg/m^3}$, nuclear matter $2\times10^{17}\un{kg/m^3}$.
  \item Speeds: sound in air $343\un{m/s}$, in water $1500\un{m/s}$; molecules in air $\sim500\un{m/s}$; escape from Earth $11.2\un{km/s}$; Earth's orbit $30\un{km/s}$.
  \item A mole of gas at room temperature and pressure occupies about $24$ litres.
  \item Viscosity of water $10^{-3}\un{Pa\,s}$; of air $1.8\times10^{-5}\un{Pa\,s}$.
\end{itemize}
